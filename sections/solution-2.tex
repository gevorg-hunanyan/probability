\begin{sol}{two-shooters}
    We assume the events $A_1$ and $A_2$, of first and second shooters hitting the target, are independent. The probability that the target was hit at least once is then
    \begin{multline*}
        \PP(\text{hit at least once})=1-\PP(\text{both miss})=1-\PP(\overline{A_1}\cap\overline{A_2})=1-\PP(\overline{A_1})\PP(\overline{A_2})\\=1-0.3\cdot0.2=1-0.06=0.94,
    \end{multline*}
    and the probability of exactly one hit is
    \begin{multline*}
        \PP(\text{exactly one hit})=\PP\left((A_1\cap\overline{A_2})\cup(\overline{A_1}\cap A_2)\right)=\PP(A_1\cap\overline{A_2})+\PP(\overline{A_1}\cap A_2)\\=\PP(A_1)\PP(\overline{A_2})+\PP(\overline{A_1})\PP(A_2)=0.7\cdot0.2+0.3\cdot0.8=0.14+0.24=0.38.
    \end{multline*}
\end{sol}

\begin{sol}{five-points-circle-square}
    Let the radius of the circle be $r$. The diagonal of an inscribed square is $2r$, side length is $\sqrt{2}r$, and the area is $S_\square=(\sqrt{2}r)^2=2r^2$. The area of the circle is $S_\bigcirc=\pi r^2$, hence
    \[
        p=\PP(\text{point in square})=\frac{2r^2}{\pi r^2}=\frac{2}{\pi}.
    \]
    The region outside the square is composed of 4 segments of equal area, that add up to $\pi r^2-2r^2$, so each one has area $\frac{\pi r^2-2r^2}{4}$. Therefore
    \[
        q=\PP(\text{point in a chosen segment})=\frac{\pi r^2-2r^2}{4\pi r^2}=\frac{\pi-2}{4\pi}.
    \]
    The desired probability is
    \[
        \binom{5}{1}\cdot p\cdot 4!\cdot q^4 = 120\cdot\frac{2}{\pi}\left(\frac{\pi-2}{4\pi}\right)^4.
    \]
\end{sol}

\begin{sol}{student-library}
    Since the probability that the student will find the book in a library is $0.5$, and the probability that the book will be occupied is $0.5$, the probability of the student obtaining the book at a library is $0.25$. Hence the probability that the student will not get the book in a library is $1-0.25=0.75$. Because the libraries are independently operated, the probability that the student does not obtain the book after visiting all three libraries is $(0.75)^3=\frac{27}{64}\approx0.42$. Therefore it is more likely that the student will get the book.
\end{sol}

\begin{sol}{ineq1}
    Without loss of generality assume $\PP(A)\le\PP(B)$. Then $\PP(A\cap B)\le\PP(A)\le\PP(B)$. We will prove the inequalities $\PP(A\cap B)-\PP(A)\PP(B)\le\frac14$ and $\PP(A)\PP(B)-\PP(A\cap B)\le\frac14$, which together will imply $\left|\PP(A\cap B)-\PP(A)\PP(B)\right|\le\frac14$.
    \[
        \PP(A\cap B)-\PP(A)\PP(B)\le\PP(A)-\PP(A)\PP(B)\le\PP(A)-\PP(A)^2\le\frac14,
    \]
    because if $x\in[0,1]$ $x-x^2\le\frac14$. The inequality $\PP(A\cap B)-\PP(A)\PP(B)\le\frac14$ is proved.
    
    To prove $\PP(A)\PP(B)-\PP(A\cap B)\le\frac14$ we will consider the cases $\PP(A)+\PP(B)\le1$ and $\PP(A)+\PP(B)\ge1$. If $\PP(A)+\PP(B)\le1$, which is the same as $\PP(B)\le1-\PP(A)$ then
    \[
        \PP(A)\PP(B)-\PP(A\cap B)\le\PP(A)\PP(B)\le\PP(A)(1-\PP(A))\le\frac14,
    \]
    and if $\PP(A)+\PP(B)\ge1$, which is the same as $1-\PP(B)\le\PP(A)$ then
    \begin{align*}
        \PP(A)\PP(B)-\PP(A\cap B)&\le\PP(A)\PP(B)-(\PP(A)+\PP(B)-1)\\
        &=(1-\PP(A))(1-\PP(B))\le\PP(A)(1-\PP(A))\le\frac14
    \end{align*}
    because $\PP(A\cap B)=\PP(A)+\PP(B)-\PP(A\cup B)\ge\PP(A)+\PP(B)-1$.
\end{sol}

\begin{sol}{bowlsinduction}
    We saw in the previous problem that the probability of removing a white ball from the third bowl does not depend on whether or not we add a ball to it from the second bowl. Perhaps we could prove by induction that it is still $\frac{3}{10}$ for $N$ bowls. An even more general statement would be to say that if each of the $N$ bowls contains $n$ white balls and $m$ black balls, and we do the procedure described in the previous problem, then the probability that we remove a white ball from the last bowl is $\frac{n}{n+m}$. Base case is obvious for a single bowl. For the induction step assume it holds for $N-1$ bowls: $\PP(W_{N-1})=\frac{n}{n+m}$. Then
    \begin{align*}
        \PP(W_{N})&=\PP(W_{N-1}\cap W_N)+\PP(B_{N-1}\cap W_N)\\
        &=\PP(W_{N-1})\PP(W_N\mid W_{N-1})+\PP(B_{N-1})\PP(W_N\mid B_{N-1})\\
        &=\frac{n}{n+m}\cdot\frac{n+1}{n+m+1}+\frac{m}{n+m}\cdot\frac{n}{n+m+1}\\
        &=\frac{n(n+1)+nm}{(n+m)(n+m+1)}=\frac{n(n+m+1)}{(n+m)(n+m+1)}=\frac{n}{n+m},
    \end{align*}
    which concludes the proof.
\end{sol}

\begin{sol}{box15balls}
    Let $X$ and $Y$ be the number of new balls drawn on the first and second days, respectively, and let $k\in\{0,1,2,3\}$. Then
    \[
        \PP(X=k)=\frac{\binom{9}{k}\binom{6}{3-k}}{\binom{15}{3}}.
    \]
    After returning these balls we will have $9-k$ new balls, so
    \[
        \PP(Y=3\mid X=k)=\frac{\binom{9-k}{3}}{\binom{15}{3}},
    \]
    and it follows from the law of total probability, that
    \[
        \PP(Y=3)=\sum_{k=0}^{3}\PP(X=k)\PP(Y=3\mid X=k)=\sum_{k=0}^{3}\frac{\binom{9}{k}\binom{6}{3-k}}{\binom{15}{3}}\cdot\frac{\binom{9-k}{3}}{\binom{15}{3}}=\frac{528}{5915}\approx9\%.
    \]
\end{sol}
